# ai_client.py - 行业检测核心逻辑
def detect_industry(name: str) -> tuple:
    name_lower = name.lower()
    
    if any(k in name_lower for k in ["银行", "保险", "平安"]):
        return ("金融服务业", ..., [...], [...])
        
    elif any(k in name_lower for k in ["新能源", "储能", "思格"]):
        return ("新能源与储能行业", ..., [...], [...])
        
    elif any(k in name_lower for k in ["制造", "万向"]):
        return ("高端制造业", ..., [...], [...])
    
    # ... 其他行业
    
    return ("现代服务业", ..., [...], [...])

# 使用 MD5 hash 保证一致性
hash_val = int(hashlib.md5(company_name.encode()).hexdigest()[:8], 16)
random.seed(hash_val)  # 同一公司每次结果相同
```

def calculate_emotion_score(quote_data):
    """
    基于真实股价数据的量化情绪计算
    
    权重分配:
    - 价格动量 (涨跌幅): 40%  ← 主要因子
    - 波动率: 20%           ← 市场稳定性
    - RSI相对强弱: 20%       ← 超买超卖
    - MA趋势判断: 20%        ← 中期方向
    """
    score = (
        price_momentum * 0.40 +
        volatility * 0.20 +
        rsi_indicator * 0.20 +
        ma_trend * 0.20
    )
    
    # 归一化到 -100 ~ +100
    normalized_score = normalize(score, -100, 100)
    
    return {
        'score': normalized_score,
        'metadata': {
            'is_real_data': True,
            'data_source': quote_data.get('source'),  # alpha_vantage / yahoo_finance
            'price': quote_data.get('price'),
            'change_pct': quote_data.get('change_percent'),
            'volume': quote_data.get('volume'),
            'update_time': datetime.now().isoformat()
        }
    }
```
